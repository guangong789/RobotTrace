cmake_minimum_required(VERSION 3.16)
project(robottrace_inference LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ROBOTTRACE_CUDA_BUILD "Build the explicitly selected CUDA Processor" OFF)
if(ROBOTTRACE_CUDA_BUILD)
    set(ORT_GPU_ROOT "/opt/rt/ort" CACHE PATH "Isolated GPU ORT SDK")
    set(ORT_ROOT "${ORT_GPU_ROOT}")
    set(pipeline_target inference_pipeline_cuda)
else()
    set(ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.deps/onnxruntime-linux-x64-1.20.1")
    set(pipeline_target inference_pipeline)
endif()
find_library(ORT_LIBRARY onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/InferenceInput.msg" "msg/InferenceResult.msg")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
foreach(target ${pipeline_target} inference_contract)
    if(target STREQUAL "${pipeline_target}")
        add_executable(${target} src/pipeline.cpp)
    else()
        add_executable(${target} ../../tests/inference_contract.cpp)
    endif()
    target_include_directories(${target} PRIVATE include ../robottrace_examples/include)
    target_include_directories(${target} PRIVATE ../../experiments/cpu_inference)
    target_include_directories(${target} SYSTEM PRIVATE "${ORT_ROOT}/include")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(${target} PRIVATE ROBOTTRACE_ORT_ROOT="${ORT_ROOT}")
    if(ROBOTTRACE_CUDA_BUILD)
        target_compile_definitions(${target} PRIVATE ROBOTTRACE_CUDA_BUILD=1)
    endif()
    ament_target_dependencies(${target} rclcpp)
    target_link_libraries(${target} "${typesupport}" "${ORT_LIBRARY}")
endforeach()
set_target_properties(${pipeline_target} PROPERTIES INSTALL_RPATH "$ORIGIN/..;${ORT_ROOT}/lib")
install(TARGETS ${pipeline_target} RUNTIME DESTINATION lib/${PROJECT_NAME})
enable_testing()
add_test(NAME inference_contract COMMAND inference_contract "${CMAKE_CURRENT_SOURCE_DIR}/../../models/mobilenet")
ament_export_dependencies(rosidl_default_runtime)
ament_package()
